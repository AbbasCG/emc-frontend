import { useState, useEffect } from 'react';
import { motion } from 'framer-motion';
import { BookOpen, Calendar, Video, Clock, Award, Bell, ChevronRight, FileText, Settings, LogOut, Headphones } from 'lucide-react';
import { EmcButton } from '@/components/ui';
import { instituteApi } from '@/api/instituteApi';
 
interface InstitutePricingOption {
  title: string;
  price: number;
}
 
interface InstituteLevel {
  id: number;
  title: string;
  description?: string | null;
  price?: number;
  pricing_options?: InstitutePricingOption[];
}
 
interface InstituteNextClass {
  topic: string;
  date: string;
  time: string;
  teacher: string;
}
 
/** Shape of GET /institute/student/dashboard, discriminated by enrollment status. */
type InstituteStudentDashboardData =
  | { status: 'not_enrolled'; available_levels?: InstituteLevel[] }
  | { status: 'waitlist'; current_level: string }
  | { status: 'enrolled'; current_level: string; progress: number; next_class: InstituteNextClass };

  const [isLoading, setIsLoading] = useState(true);
 
  useEffect(() => {
    const controller = new AbortController();
    void (async () => {
      try {
        const res = await instituteApi.getStudentDashboard();
        if (controller.signal.aborted) return;
        setDashboardData(res.data as InstituteStudentDashboardData);
      } catch (err) {
        if (controller.signal.aborted) return;
        if (import.meta.env.DEV) console.error(err);
      } finally {
        if (!controller.signal.aborted) setIsLoading(false);
